cmake_minimum_required(VERSION 3.20)
project(tinylsm VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(Threads REQUIRED)
option(TINYLSM_ASAN "Enable address and undefined sanitizers" OFF)
option(TINYLSM_TSAN "Enable thread sanitizer" OFF)
option(TINYLSM_COVERAGE "Enable gcov coverage" OFF)
if(TINYLSM_ASAN AND TINYLSM_TSAN)
  message(FATAL_ERROR "ASan and TSan require separate builds")
endif()
if(TINYLSM_ASAN)
  add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
  add_link_options(-fsanitize=address,undefined)
endif()
if(TINYLSM_TSAN)
  add_compile_options(-fsanitize=thread -fno-omit-frame-pointer)
  add_link_options(-fsanitize=thread)
endif()
if(TINYLSM_COVERAGE)
  add_compile_options(--coverage -O0 -g)
  add_link_options(--coverage)
endif()
add_library(tinylsm src/db.cpp src/table.cpp src/workers.cpp src/recovery.cpp)
target_include_directories(tinylsm PUBLIC include)
target_link_libraries(tinylsm PUBLIC Threads::Threads)
target_compile_options(tinylsm PRIVATE -Wall -Wextra -Wpedantic)
enable_testing()
add_executable(tinylsm_tests tests/test.cpp)
target_include_directories(tinylsm_tests PRIVATE src)
target_link_libraries(tinylsm_tests PRIVATE tinylsm)
add_test(NAME tinylsm_tests COMMAND tinylsm_tests)
add_executable(tinylsm_table_tests tests/table_test.cpp)
target_include_directories(tinylsm_table_tests PRIVATE src)
target_link_libraries(tinylsm_table_tests PRIVATE tinylsm)
add_test(NAME tinylsm_table_tests COMMAND tinylsm_table_tests)
add_executable(tinylsm_compaction_repro tests/compaction_repro.cpp)
target_link_libraries(tinylsm_compaction_repro PRIVATE tinylsm)
add_test(NAME tinylsm_compaction_repro COMMAND tinylsm_compaction_repro)
add_test(NAME tinylsm_sst_lifecycle COMMAND tinylsm_tests --sst)
include(tests/recovery.cmake)
include(scripts/tools.cmake)
add_executable(tinylsm_fault_injection tests/fault_injection.cpp)
target_include_directories(tinylsm_fault_injection PRIVATE src)
target_link_libraries(tinylsm_fault_injection PRIVATE tinylsm)
add_test(NAME tinylsm_fault_injection COMMAND tinylsm_fault_injection)
set_tests_properties(tinylsm_fault_injection PROPERTIES TIMEOUT 180)
add_executable(tinylsm_compaction_levels tests/compaction_levels.cpp)
target_link_libraries(tinylsm_compaction_levels PRIVATE tinylsm)
add_test(NAME tinylsm_compaction_levels COMMAND tinylsm_compaction_levels)
set_tests_properties(tinylsm_compaction_levels PROPERTIES TIMEOUT 120)
set_tests_properties(tinylsm_tests tinylsm_table_tests tinylsm_compaction_repro tinylsm_sst_lifecycle PROPERTIES TIMEOUT 20)
